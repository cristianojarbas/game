        const formatCurrency = (amount) => new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', minimumFractionDigits: 0 }).format(amount);

        const calculateMandI = (principal, annualRate, years) => {

            const r = annualRate / 12;

            const n = years * 12;

            if (r === 0) return principal / n;

            return principal * (r * Math.pow(1 + r, n)) / (Math.pow(1 + r, n) - 1);

        };

        const callGeminiApi = async (userQuery, systemPrompt, useGrounding = false) => {

            const payload = {

                systemInstruction: { parts: [{ text: systemPrompt }] },

            };

                        method: 'POST',

                        headers: { 'Content-Type': 'application/json' },

                        body: JSON.stringify(payload)

                    });


                    if (!response.ok) {

                        if (response.status === 429 && attempt < maxRetries - 1) {

                            const delay = Math.pow(2, attempt) * 1000 + Math.random() * 1000;

                            await new Promise(resolve => setTimeout(resolve, delay));

                            continue;

                        }

                        throw new Error(`API call failed with status: ${response.status}`);

                    }


                    const result = await response.json();

                    const candidate = result.candidates?.[0];

                        return candidate.content.parts[0].text;

                    } else {

                        throw new Error("Invalid response structure from Gemini API.");

                    }


                } catch (error) {

                    if (attempt === maxRetries - 1) {

                        console.error("Gemini API max retries reached.", error);

                        throw new Error("Failed to get advice from the AI advisor. Please try again later.");

        const renderPreApproval = () => {

            const currentTotalDebt = Object.values(gameState.currentDebts).reduce((a, b) => a + b, 0);

            const dti = (currentTotalDebt / gameState.income).toFixed(3);

            const isDTIok = dti <= MAX_DTI;


            if (!isDTIok) {

                 return `<h3 class="text-2xl font-bold text-red-500">DTI Failure!</h3><p class="text-gray-300">Your DTI is too high. Return to **Savings Surge** to pay down debt.</p><button onclick="renderCenterContent('1-2')" class="btn-primary mt-4">Go Back to DTI</button>`;

            }


            const maxPITI = gameState.income * MAX_DTI;

            const maxMandI = maxPITI - currentTotalDebt - (INSURANCE_COST / 12);

            

            const marketData = MARKET_SCENARIOS[gameState.market];

            const interestRate = marketData.interestRate;

            const dp = gameState.downPaymentPct;


            const r = interestRate / 12;

            const n = 360; 

            let maxLoanAmount = maxMandI * (Math.pow(1 + r, n) - 1) / (r * Math.pow(1 + r, n));

            

            if (maxLoanAmount <= 50000) {

                 showModal('Loan Denied', 'Your financial profile only allows a very small loan. Try paying down more debt or increasing your simulated income next time!');

            }


            const maxOfferPrice = roundToNearest(maxLoanAmount / (1 - dp), 1000);

            maxLoanAmount = roundToNearest(maxOfferPrice * (1 - dp), 1000);

                <h3 class="text-2xl font-bold text-secondary mb-4">PRE-APPROVED!</h3>

                <p class="text-sm text-gray-400 mt-4">With your ${formatPercent(dp)} down payment, you're ready to shop. Click **Search Listings** to continue!</p>

                <button onclick="advanceStage()" class="btn-primary mt-6">Start House Hunting</button>

            `;

        };

        // --- STAGE 2: HOUSE HUNTING ---

        

        const generateListings = (maxPrice) => {

            const listings = [];

            const marketData = MARKET_SCENARIOS[gameState.market];

            const [min, max] = marketData.priceRange;


            for (let i = 0; i < 5; i++) {

                let price = roundToNearest(Math.random() * (max - min) + min, 5000);

                price = Math.min(price, maxPrice * 1.05); 


                const loanAmt = price * (1 - gameState.downPaymentPct);

                const pmi = (gameState.downPaymentPct < 0.20) ? loanAmt * PMI_RATE / 12 : 0;

                const taxes = price * marketData.taxRate / 12;

                const insurance = INSURANCE_COST / 12;

                const mandI = calculateMandI(loanAmt, marketData.interestRate, 30);

                const piti = mandI + taxes + insurance + pmi;


                listings.push({

                    id: `prop${i + 1}`,

                    price: price,

                    beds: Math.floor(Math.random() * 3) + 2,

                    baths: (Math.floor(Math.random() * 2) + 1) + (Math.random() > 0.5 ? 0.5 : 0),

                    sqft: roundToNearest(Math.random() * 1000 + 1500, 50),

                    piti: piti,

                    taxes: taxes,

                    condition: Math.random() > 0.7 ? 'Excellent' : (Math.random() > 0.5 ? 'Good' : 'Needs Work'),

                    redFlags: Math.floor(Math.random() * 4) + 1, 

                });

            }

            return listings.filter(l => l.price <= maxPrice * 1.05);

        };

            return `

                <h3 class="text-xl font-bold text-primary mb-4">Search Listings</h3>

                <p class="text-gray-300 mb-4">Here are 5 potential homes. Your max budget is <span class="font-bold text-accent">${formatCurrency(gameState.maxOfferPrice)}</span>.</p>

                

                <div class="max-h-60 overflow-y-auto space-y-3 p-2 bg-gray-700 rounded-lg">

                    ${currentListings.map(l => `

                        <div class="bg-gray-800 p-3 rounded-lg flex justify-between items-center text-left">

                <button onclick="advanceStage()" class="btn-primary mt-6">Next: Comps Corner</button>

                <h3 class="text-xl font-bold text-primary mb-4">Comps Corner (Market Analysis)</h3>

                <p class="text-gray-300 mb-4">Analyze the 5 potential properties you found. Click **Select Property** when you find your favorite.</p>

                <div class="max-h-60 overflow-y-auto space-y-3 p-2 bg-gray-700 rounded-lg">

                    ${currentListings.map(l => `

                        <div class="bg-gray-800 p-3 rounded-lg flex justify-between items-center text-left">

                            <div>

                                <h4 class="font-bold text-secondary">${formatCurrency(l.price)}</h4>

        const selectProperty = (id) => {

            const property = currentListings.find(l => l.id === id);

            if (!property) return;

            

            const inspection = { critical: [], nonCritical: [] };

            for(let i = 0; i < property.redFlags; i++) {

                inspection.critical.push(`CRITICAL: Foundation/Roof issue requiring ${formatCurrency(roundToNearest(Math.random() * 15000 + 5000, 100))}`);

            }

            inspection.nonCritical.push(`NON-CRITICAL: Loose faucet in bathroom.`);

            

            saveGameState({ property, inspectionResult: inspection });

            renderCenterContent('2-3'); // Re-render the selection space

        };

        window.selectProperty = selectProperty;


        const renderPropertySelection = () => {

            if (!gameState.property) {

                return `<h3 class="text-xl text-yellow-500">No Property Selected.</h3><p class="text-gray-300">Go back to **Comps Corner** to select a listing.</p><button onclick="renderCenterContent('2-2')" class="btn-primary mt-4">Go Back</button>`;

            }

            const prop = gameState.property;

            const dpAmount = prop.price * gameState.downPaymentPct;


            return `

                <h3 class="text-xl font-bold text-primary">Selected Home: ${formatCurrency(prop.price)}</h3>

                <div class="bg-gray-700 p-4 rounded-lg space-y-2 my-4 text-left">

                    <p>Est. PITI: <span class="font-bold text-secondary">${formatCurrency(prop.piti)}</span></p>

                    <p>Est. Down Payment (${formatPercent(gameState.downPaymentPct)}): <span class="font-bold">${formatCurrency(dpAmount)}</span></p>

                    <p>Condition: <span class="font-bold text-accent">${prop.condition}</span></p>

                </div>

                <p class="text-sm text-gray-400">Click **Make Offer** to finalize your bid.</p>

                <button onclick="advanceStage()" class="btn-primary mt-6">Next: Make Offer</button>

            `;

        };


        const renderOfferBuilder = () => {

            if (!gameState.property) return `<h3 class="text-xl text-red-500">Error: No Property Data</h3>`;

            const prop = gameState.property;


            return `

                <h3 class="text-xl font-bold text-primary">Crafting Offer on ${formatCurrency(prop.price)}</h3>

                <div class="text-left text-gray-200 space-y-4 pt-4">

                    <label class="block font-semibold">Offer Price (Max ${formatCurrency(gameState.maxOfferPrice)})</label>

                    <input type="number" id="offer-price" value="${prop.price}" min="${prop.price * 0.9}" max="${gameState.maxOfferPrice}" class="w-full p-2 border rounded-lg bg-gray-700 text-white">


                    <label class="block font-semibold">Earnest Money Deposit (EMD)</label>

                    <input type="number" id="emd-amount" value="5000" min="1000" step="1000" class="w-full p-2 border rounded-lg bg-gray-700 text-white">


                    <p class="font-semibold pt-2">Contingencies:</p>

                    <div class="space-y-1 text-sm">

                        <label class="flex items-center space-x-2"><input type="checkbox" id="cont-inspection" checked class="form-checkbox text-secondary rounded"><span>Inspection</span></label>

                        <label class="flex items-center space-x-2"><input type="checkbox" id="cont-financing" checked class="form-checkbox text-secondary rounded"><span>Financing</span></label>

                        <label class="flex items-center space-x-2"><input type="checkbox" id="cont-appraisal" checked class="form-checkbox text-secondary rounded"><span>Appraisal</span></label>

                    </div>

                </div>

                <button onclick="submitOffer()" class="btn-primary w-full mt-6">Submit Offer</button>

            `;

        };

        window.renderOfferBuilder = renderOfferBuilder; 

        

        const submitOffer = () => {

            const offerPrice = parseInt(document.getElementById('offer-price').value);

            const emd = parseInt(document.getElementById('emd-amount').value);

            const contingencies = {

                inspection: document.getElementById('cont-inspection').checked,

                financing: document.getElementById('cont-financing').checked,

                appraisal: document.getElementById('cont-appraisal').checked,

            };


            if (offerPrice > gameState.maxOfferPrice) {

                showModal('Offer Too High!', `Your offer of ${formatCurrency(offerPrice)} exceeds your max pre-approved amount of ${formatCurrency(gameState.maxOfferPrice)}.`);

                return;

            }


            const marketData = MARKET_SCENARIOS[gameState.market];

            let acceptanceChance = 0.5 + (emd / 10000) * 0.2; 

            if (offerPrice >= gameState.property.price * 1.05) acceptanceChance += marketData.volatility;

            if (!contingencies.appraisal || !contingencies.inspection) acceptanceChance += 0.15;


            if (Math.random() < acceptanceChance) {

                showModal('OFFER ACCEPTED!', `<p>Your offer of **${formatCurrency(offerPrice)}** has been accepted! Click **Inspection Report** to start due diligence.</p>`);

                advanceStage({ offer: { price: offerPrice, emd, contingencies } });

            } else {

                showModal('Offer Rejected', `<p>Unfortunately, your offer was rejected. Try increasing your price or reducing contingencies.</p>`);

                renderCenterContent('2-4'); 

            }

        };

        window.submitOffer = submitOffer;



        // --- STAGE 3: DUE DILIGENCE & CLOSING ---


        const renderInspectionReport = () => {

            const criticalIssues = gameState.inspectionResult.critical;

            return `

                <h3 class="text-xl font-bold text-red-500 mb-3">Inspection Report: CRITICAL DEFECTS</h3>

                <button onclick="generateNegotiationScript(${JSON.stringify(criticalIssues)})" class="btn-llm mb-4">

                    ✨ Draft Negotiation Script

                </button>

                <div class="max-h-40 overflow-y-auto bg-gray-700 p-3 rounded-lg text-left text-gray-200 space-y-1">

                    ${criticalIssues.map(issue => `<p class="text-sm font-semibold text-red-300">- ${issue}</p>`).join('')}

                    <p class="text-xs text-gray-400 pt-2">...and other non-critical items found.</p>

                </div>


                <p class="font-bold text-gray-300 mt-4">Negotiation Decision:</p>

                <button onclick="handleNegotiation('credit')" class="btn-primary mr-3 bg-blue-600 hover:bg-blue-700 text-sm">Request Closing Credit</button>

                <button onclick="handleNegotiation('accept')" class="btn-primary bg-secondary text-sm">Accept As-Is</button>

                <button onclick="handleNegotiation('walk')" class="btn-primary bg-red-600 hover:bg-red-700 mt-4 text-sm">Walk Away (Lose EMD)</button>

            `;

        };

        window.handleNegotiation = (action) => {

            if (action === 'walk') {

                const newSavings = gameState.savings - gameState.offer.emd;

                showModal('Contract Cancelled', `You walked away and lost your **EMD** of ${formatCurrency(gameState.offer.emd)}.`);

                saveGameState({ stage: '2-1', savings: newSavings, property: null, offer: null });

                return;

            }


            const sellerAgrees = Math.random() > 0.3;


            if (sellerAgrees || action === 'accept') {

                let message = "Contingency Cleared! The seller accepted your terms.";

                let newSavings = gameState.savings;

                if (action === 'credit') {

                    const creditAmount = roundToNearest(Math.random() * 5000 + 2000, 100);

                    newSavings += creditAmount;

                    message = `The seller granted a **${formatCurrency(creditAmount)} closing credit**. Savings updated!`;

                }


                showModal('Inspection Cleared', message);

                advanceStage({ savings: newSavings, closingSubStage: 'appraisal' });

            } else {

                 showModal('Negotiation Failed', 'The seller is unwilling to budge. You must accept as-is or walk away.',

                    `<button onclick="hideModal(); handleNegotiation('accept')" class="btn-primary mr-3 text-sm">Accept As-Is</button>

                     <button onclick="hideModal(); handleNegotiation('walk')" class="btn-primary bg-red-600 text-sm">Walk Away</button>`);

            }

        };



        const renderAppraisal = () => {

            const appraisalBase = gameState.property.price * 0.95;

            const appraisalVariance = gameState.property.price * MARKET_SCENARIOS[gameState.market].volatility;

            const finalAppraisal = roundToNearest(Math.random() * appraisalVariance + appraisalBase, 1000);

            const gap = gameState.offer.price - finalAppraisal;

            

            saveGameState({ appraisalValue: finalAppraisal }); 

            

            let html;

            if (gap <= 0) {

                html = `<h3 class="text-2xl font-bold text-secondary mb-4">Appraisal Success!</h3>

                        <p class="text-gray-300">Appraisal Value: **${formatCurrency(finalAppraisal)}**. You are safe!</p>

                        <button onclick="advanceStage()" class="btn-primary mt-6">Next: Closing Disclosure</button>`;

            } else {

                html = `<h3 class="text-2xl font-bold text-red-500 mb-4">APPRAISAL GAP: ${formatCurrency(gap)}</h3>

                        <p class="text-gray-300">The home appraised for only **${formatCurrency(finalAppraisal)}**.</p>

                        <p class="font-bold mt-2 text-accent">You must cover the ${formatCurrency(gap)} difference with cash.</p>

                        ${gameState.offer.contingencies.appraisal ? `

                            <button onclick="handleAppraisalGap('cash', ${gap})" class="btn-primary mr-3 bg-blue-600 mt-6">Bring Cash to Close</button>

                            <button onclick="handleAppraisalGap('walk', ${gap})" class="btn-primary bg-red-600 mt-6">Walk Away (No Loss)</button>

                        ` : `

                            <p class="text-red-500 font-bold mt-3">You WAIVED the Contingency!</p>

                            <button onclick="handleAppraisalGap('cash', ${gap})" class="btn-primary mr-3 bg-blue-600 mt-6">Bring Cash to Close</button>

                            <button onclick="handleAppraisalGap('lose_emd', ${gap})" class="btn-primary bg-red-600 mt-6">Walk Away (Lose EMD)</button>

                        `}

            `;

            }

            return html;

        };

        window.handleAppraisalGap = (action, gap) => {

            if (action === 'cash') {

                const newSavings = gameState.savings - gap;

                if (newSavings < 0) {

                     showModal('FATAL ERROR', `Savings too low! Game Over.`);

                     resetGame();

                     return;

                }

                showModal('Gap Covered', `You brought **${formatCurrency(gap)}** in cash. Savings updated.`);

                advanceStage({ savings: newSavings });

            } else if (action === 'walk') {

                showModal('Contract Cancelled', `You walked away due to the appraisal gap. You lose nothing.`);

                saveGameState({ stage: '2-1', property: null, offer: null });

            } else if (action === 'lose_emd') {

                const newSavings = gameState.savings - gameState.offer.emd;

                showModal('Contract Cancelled', `You lost your EMD of ${formatCurrency(gameState.offer.emd)}.`);

                saveGameState({ stage: '2-1', savings: newSavings, property: null, offer: null });

            }

        };



        const renderClosingDisclosure = () => {

            const loanAmount = gameState.property.price * (1 - gameState.downPaymentPct);

            const closingCostPercent = 0.035; 

            const appraisalGap = Math.max(0, gameState.offer.price - gameState.appraisalValue);

            const totalCashNeeded = roundToNearest(gameState.property.price * gameState.downPaymentPct + loanAmount * closingCostPercent + appraisalGap, 100);

            

            saveGameState({ cashToClose: totalCashNeeded });


            return `

                <h3 class="text-xl font-bold text-primary mb-4">Closing Disclosure Summary</h3>

                <div class="bg-gray-700 p-4 rounded-lg space-y-2 text-left text-gray-200">

                    <div class="flex justify-between"><span>Down Payment:</span> <span class="font-bold">${formatCurrency(gameState.property.price * gameState.downPaymentPct)}</span></div>

                    <div class="flex justify-between"><span>Closing Costs (Est.):</span> <span class="font-bold">${formatCurrency(loanAmount * closingCostPercent)}</span></div>

                    ${appraisalGap > 0 ? `<div class="flex justify-between text-red-400"><span>Appraisal Gap Coverage:</span> <span class="font-bold">${formatCurrency(appraisalGap)}</span></div>` : ''}

                    <hr class="my-2 border-gray-500">

                    <div class="flex justify-between text-lg font-black text-accent"><span>TOTAL CASH TO CLOSE:</span> <span>${formatCurrency(totalCashNeeded)}</span></div>

                </div>


                <p class="mt-6 text-gray-300">Your Current Savings: <span class="font-bold text-accent text-lg">${formatCurrency(gameState.savings)}</span></p>

                

                ${gameState.savings >= totalCashNeeded ? `

                    <button onclick="advanceStage()" class="btn-primary w-full mt-6">Next: Closing Day</button>

                ` : `

                    <p class="text-red-500 font-bold text-center mt-6">CRITICAL: Insufficient Funds!</p>

                    <button onclick="resetGame()" class="btn-primary bg-red-600 w-full mt-4">Game Over: Foreclosure</button>

                `}

            `;

        };


        const renderClosingDay = () => {

             const newSavings = gameState.savings - gameState.cashToClose;

             saveGameState({ savings: newSavings });

            return `

                <h3 class="text-2xl font-bold text-secondary mb-4">Signing Day!</h3>

                <p class="text-gray-300">You signed the final documents! You paid **${formatCurrency(gameState.cashToClose)}** in cash to close.</p>

                <button onclick="advanceStage()" class="btn-primary mt-6">See Final Summary</button>

            `;

        };

        

        const renderCompletionScreen = () => {

            return `

                <h2 class="text-3xl font-black text-accent mb-4">🎉 CONGRATULATIONS! 🎉</h2>

                <h3 class="text-xl font-bold text-secondary mb-6">You are now a Homeowner!</h3>

                    <p>Property Price: **${formatCurrency(gameState.property.price)}**</p>

                    <p>Total Cash Invested: **${formatCurrency(gameState.cashToClose)}**</p>

                    <p class="mt-3 text-lg font-bold text-accent">Remaining Savings: ${formatCurrency(gameState.savings)}</p>
